```typescript
// lib/db/client.ts
export const sql: NeonQueryFunction; // @neondatabase/serverless, neon(process.env.DATABASE_URL!) — HTTP, para queries sueltas de lectura
export function withTx<T>(fn: (client: PoolClient) => Promise<T>): Promise<T>;
// withTx abre un `Pool` de @neondatabase/serverless (WebSocket, no el driver HTTP: el HTTP `sql.transaction()`
// solo manda un batch pre-armado y no puede branchear sobre una lectura intermedia). withTx hace
// client.connect() -> BEGIN -> fn(client) -> COMMIT (o ROLLBACK si fn tira), y libera el client al final.
// El withdraw (lock -> sumar 24h -> rechazar/debitar) y el interés (leer delta -> if/else) necesitan
// justamente eso: una sesión interactiva real, no un batch.

// lib/privy-server.ts
export interface PrivyIdentity { userId: string; walletAddress: `0x${string}` | null }
export async function verifyPrivyToken(authHeader: string | null): Promise<PrivyIdentity>; // throws 401 si inválido
```

Formato de la tabla `movements.type`: `'deposit' | 'withdraw' | 'interest'`, CHECK constraint. Balances en `numeric(38,0)`, siempre enteros (unidades base, 18 decimals), nunca floats en el código TS: usar `bigint`. Postgres devuelve `numeric` como **string** en JS: todo valor leído de `argt_balance`/`bolt_balance`/`amount` se pasa por `BigInt(valor)` antes de cualquier suma, resta o comparación — nunca operar sobre el string ni castear a `number`.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| Browser → `/api/cuenta/*` | Access token de Privy en el header `Authorization`, único mecanismo de auth de estas rutas |
| API routes → Postgres (Neon) | `DATABASE_URL` server-only, nunca expuesta al cliente |
| API routes → Privy API | `PRIVY_APP_SECRET` server-only, usado para verificar tokens y resolver la embedded wallet |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|------------------|
| T-03-01 | Spoofing | `app/api/cuenta/account/route.ts` | mitigate | `verifyPrivyToken` valida el access token contra Privy server-side antes de resolver `userId`; nunca se confía en un `userId` mandado por el cliente |
| T-03-02 | Tampering | `lib/db/seed.ts` | accept | Script de seed solo corre manualmente contra la DB de la demo, no es un endpoint expuesto |
| T-03-03 | Information Disclosure | `DATABASE_URL`, `PRIVY_APP_SECRET` | mitigate | Variables server-only (sin prefijo `NEXT_PUBLIC_`), nunca referenciadas desde componentes cliente |
</threat_model>
